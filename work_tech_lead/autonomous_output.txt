### `strategies/moving_average_strategy.py`
```python
from models.trading_model import TradingModel

class MovingAverageStrategy:
    def __init__(self, short_window=50, long_window=200):
        self.short_window = short_window
        self.long_window = long_window
        self.model = TradingModel()

    def execute(self, data):
        short_avg = data['close'].rolling(window=self.short_window).mean()
        long_avg = data['close'].rolling(window=self.long_window).mean()

        if short_avg[-1] > long_avg[-1]:
            signal = 'buy'
        elif short_avg[-1] < long_avg[-1]:
            signal = 'sell'
        else:
            signal = self.model.predict(data)

        return signal
```

### `main.py`
```python
from config.settings import TRADE_SYMBOL, TIME_FRAME
from strategies.moving_average_strategy import MovingAverageStrategy
from utils.market_utils import fetch_data, execute_trade

def main():
    data = fetch_data(TRADE_SYMBOL, TIME_FRAME)
    strategy = MovingAverageStrategy()
    signal = strategy.execute(data)

    if signal:
        execute_trade(signal, 1)  # Execute trade with 1 unit

if __name__ == "__main__":
    main()
```
